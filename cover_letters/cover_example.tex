\documentclass[]{cover}

% Override cover.cls's default Lato/Raleway fonts with Times New Roman for
% this letter. cover.cls's commands (\namesection, \lettercontent, \closing,
% \signature, \currentdate) each hardcode their own \fontspec call, so
% \setmainfont/\setsansfont alone would not reach them - the commands are
% redefined below to use \coverletterfont instead. BoldFont points at the
% times-bold.ttf companion so \bfseries/\textbf (name, itemize labels) render
% a true bold face instead of a synthesized/absent-shape substitution.
\newcommand{\coverletterfont}{\fontspec[Path = OpenFonts/fonts/times-new-roman/, BoldFont = times-bold]{times}}
\setmainfont[Path = OpenFonts/fonts/times-new-roman/, BoldFont = times-bold]{times}
\setsansfont[Path = OpenFonts/fonts/times-new-roman/, BoldFont = times-bold]{times}

% Name/contact sizes (34pt/36pt name, 11pt/13pt contact line) match
% cv/main_example.tex's \namefont and \addressfont exactly, so the CV and
% cover letter headers render identically. The name text must NOT be wrapped
% in \Huge (or any other size-changing command) at the call site below - that
% would override the \fontsize{34pt}{36pt} set here.
\renewcommand{\namesection}[3]{
\centering{
\fontsize{34pt}{36pt}
\coverletterfont\selectfont\bfseries #1
\coverletterfont\selectfont\bfseries #2
} \\[5pt]
\centering{
\color{headings}
\coverletterfont\fontsize{11pt}{13pt}\selectfont #3}
\noindent\makebox[\linewidth]{\color{headings}\rule{\paperwidth}{0.0pt}}
\vspace{0pt}
}

\renewcommand{\currentdate}[1]{\raggedleft\coverletterfont\fontsize{11pt}{13pt}\selectfont {#1 \\} \normalfont}

\renewcommand{\lettercontent}[1]{\raggedright\coverletterfont\fontsize{11pt}{13pt}\selectfont {#1 \\}\mbox{}\\ \normalfont}

\renewcommand{\closing}[1]{\raggedright\coverletterfont\fontsize{11pt}{13pt}\selectfont {#1 \\}\mbox{}\\\mbox{}\\ \normalfont}

\renewcommand{\signature}[1]{\raggedright\coverletterfont\fontsize{11pt}{13pt}\selectfont {#1 \\} \normalfont}

\currentdate{\today}
\lettercontent{Dear [Hiring Manager / Team],}

\lettercontent{[Opening paragraph: name the role and where you found it, state your strongest connection to it in one sentence, and preview why you are a fit. Keep it to 2--3 sentences.]}

\lettercontent{[Body paragraph: your most relevant experience, framed toward the tasks in the posting. Follow with 3--5 concrete bullets:]}

{\raggedright\coverletterfont\fontsize{11pt}{13pt}\selectfont
\begin{itemize}
    \item \textbf{[Achievement 1]:} [concrete result with a number where possible]
    \item \textbf{[Achievement 2]:} [skill or project mapped to a posting requirement]
    \item \textbf{[Achievement 3]:} [evidence for a nice-to-have requirement]
\end{itemize}\par}
\vspace{6pt}

\lettercontent{[Connection paragraph: why this company specifically. Reference a verified specific: a product, a stated priority, a team. Never generic.]}

\lettercontent{[Personal fit paragraph: behavioral strengths and what you bring to the team, 2--3 sentences.]}

\lettercontent{I look forward to hearing from you.}

\begin{flushright}
% Note: no trailing \\ inside \closing{} - cover.cls appends its own \\, and a
% doubled break triggers "There's no line here to end."
\closing{Kind regards,}

\signature{[YOUR NAME]}
\end{flushright}
